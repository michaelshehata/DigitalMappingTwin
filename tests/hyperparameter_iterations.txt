-- RF-1 (Balanced Baseline):

    n_estimators=150,
    max_depth=20,
    class_weight={0:1, 1:5},
    n_jobs=-1,
    random_state=6001
)

-- RF-2 (Deeper Trees)

model = RandomForestClassifier(
    n_estimators=200,
    max_depth=30,
    class_weight={0:1, 1:7},
    n_jobs=-1,
    random_state=6001
)

-- RF-3 (More Conservative)

model = RandomForestClassifier(
    n_estimators=300,
    max_depth=15,
    class_weight="balanced",
    n_jobs=-1,
    random_state=6001
)

Iteration	Purpose
RF-1	balanced baseline
RF-2	aggressive minority detection
RF-3	generalisation + less overfitting
